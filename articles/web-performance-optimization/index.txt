    // Performance monitoring setup
    export function initPerformanceMonitoring() {
      // Track Core Web Vitals
      new PerformanceObserver((entryList) => {
        for (const entry of entryList.getEntries()) {
          // Report to analytics
          reportToAnalytics({
            metric: entry.name,
            value: entry.value,
            rating: entry.rating
          });
        }
      }).observe({ entryTypes: ['web-vitals'] });
    }
